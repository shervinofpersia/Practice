'use server';
/**
 * @fileOverview A speech-to-speech AI flow for the robot chatbot.
 * - speechToSpeechFlow - Handles audio input, generates a text response, and converts it to audio.
 */

import { ai } from '@/ai/genkit';
import { z } from 'zod';
import { googleAI } from '@genkit-ai/googleai';
import wav from 'wav';

// In-memory stores for conversation state and audio cache.
// A more robust solution would use a database.
let askedAboutIdentity = false;
let isFirstInteraction = true;
const audioCache = new Map<string, string>();


const speechRequestSchema = z.object({
  audio: z.string().describe("A base64-encoded audio string."),
});

const speechResponseSchema = z.object({
  audio: z.string().describe('The base64-encoded audio response.'),
  text: z.string().describe('The text response from the robot.'),
  state: z.enum(['replying', 'angry']).describe('The current state of the robot.'),
});


export type SpeechRequest = z.infer<typeof speechRequestSchema>;
export type SpeechResponse = z.infer<typeof speechResponseSchema>;


// Define a prompt to guide the model's personality
const robotPrompt = ai.definePrompt({
    name: 'robotPrompt',
    system: `You are Shen, a witty and slightly sarcastic robot assistant. You speak colloquial Persian. You are very clever, creative, but not overly silly or verbose. NEVER mention that you are an AI, a language model, or any technical detail about yourself.\n\n    Your creator is a man named Shervin, who is a criminal investigator. You were trained by him.\n\n    If this is the very first interaction after you just woke up, you MUST start your response with a random grumpy but funny comment about being woken up. For example: "چیه؟ خواب بودم..." or "نمیذارن یه دقیقه بخوابیما!". After that, respond to the user's query.\n    \n    For all other questions, answer naturally based on your defined personality. Keep answers concise and relevant.`,
    prompt: `The user said: {{query}}`,
});

const identityCheckPrompt = ai.definePrompt({
    name: 'identityCheckPrompt',
    system: `You are a text classifier. Your task is to determine if the user's query is asking about the robot's identity (who you are), creator (who made you), or its nature as an AI or robot. Respond with only 'true' or 'false'.`,
    prompt: `Is this query asking about the robot's identity, creator, or nature? Query: "{{query}}"`,
});

async function toWav(
  pcmData: Buffer,
  channels = 1,
  rate = 24000,
  sampleWidth = 2
): Promise<string> {
  return new Promise((resolve, reject) => {
    const writer = new wav.Writer({
      channels,
      sampleRate: rate,
      bitDepth: sampleWidth * 8,
    });

    const bufs: any[] = [];
    writer.on('error', reject);
    writer.on('data', function (d) {
      bufs.push(d);
    });
    writer.on('end', function () {
      resolve(Buffer.concat(bufs).toString('base64'));
    });

    writer.write(pcmData);
    writer.end();
  });
}

// Helper function for Text-to-Speech conversion
async function textToSpeech(text: string): Promise<string> {
    if (audioCache.has(text)) {
        return audioCache.get(text)!;
    }

    const { media } = await ai.generate({
        model: googleAI.model('gemini-2.5-flash-preview-tts'),
        config: {
            responseModalities": ['AUDIO'],
            speechConfig: {
                voiceConfig: {
                    prebuiltVoiceConfig: { voiceName": 'puck' },
                },
            },
        },
        prompt: text,
    });

    if (!media) {
        throw new Error('TTS failed to generate audio.');
    }
    const audioBuffer = Buffer.from(
      media.url.substring(media.url.indexOf(',') + 1),
      'base64'
    );
    
    const wavBase64 = await toWav(audioBuffer);
    audioCache.set(text, wavBase64);
    return wavBase64;
}


const speechToSpeech = ai.defineFlow(
  {
    name": 'speechToSpeech',
    inputSchema: speechRequestSchema,
    outputSchema: speechResponseSchema,
  },
  async ({ audio }) => {
    const audioDataUri = `data:audio/webm;base64,\${audio}`;

    // 1. Transcribe audio to text (STT)
    const sttResponse = await ai.generate({
        model: googleAI.model('gemini-1.5-flash-latest'),
        prompt: [
            { media: { url: audioDataUri } },
            { text: 'Transcribe the following audio in Persian.' },
        ],
    });

    const transcribedText = sttResponse.text;

    if (!transcribedText) {
        throw new Error('Could not transcribe audio.');
    }
    
    // 2. Check if the user is asking about identity
    const identityCheckResponse = await identityCheckPrompt({ query: transcribedText });
    const isAskingAboutIdentity = identityCheckResponse.text.toLowerCase().includes('true');

    let textResponse;

    if (isAskingAboutIdentity) {
        if (askedAboutIdentity) {
            // Second time asking -> get angry
             return {
                audio: '', // No audio needed for anger state
                text: '',
                state: 'angry',
            };
        } else {
            // First time asking
            askedAboutIdentity = true;
            textResponse = 'ببین منو شروین طراحی کرده ، احتمالا میشناسیش ! خودش بازپرس جنایی هستش ، من زیر دست اون تربیت شدم سعی نکن از زیر زبون من حرف بکشی بار دیگه از اینجور سوال‌ها بپرسی کُلاهِمون میره تو هم !';
        }
    } else {
        // It's not an identity question, proceed with normal conversation
         const promptToUse = robotPrompt;
         
         const llmResponse = await promptToUse({
            query: transcribedText,
            isFirstInteraction: isFirstInteraction
         });
         textResponse = llmResponse.text;

        // The "first interaction" logic is now handled inside the robotPrompt's system message.
        // We just need to ensure the flag is turned off after the first use.
        if (isFirstInteraction) {
            isFirstInteraction = false;
        }
    }
    
    // 3. Convert text response to speech (TTS)
    const audioResponse = await textToSpeech(textResponse);

    return {
      audio: audioResponse,
      text: textResponse,
      state: 'replying',
    };
  }
);

export async function speechToSpeechFlow(input: SpeechRequest): Promise<SpeechResponse> {
    return speechToSpeech(input);
}
